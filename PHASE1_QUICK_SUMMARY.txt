HOW TO USE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. IMPORT COMPONENTS
   import HealthProfile from './components/HealthProfile';
   import LabResultsTracker from './components/LabResultsTracker';
   import StructuredDataDashboard from './components/StructuredDataDashboard';

2. USE IN YOUR APP
   <HealthProfile userId={userId} />
   <LabResultsTracker userId={userId} />
   <StructuredDataDashboard userId={userId} />

3. IMPORT UTILITIES
   import { validateLabResult } from './utils/healthDataValidation';
   import { labResultManager } from './utils/firestoreHealthDataManager';

4. USE VALIDATION
   const validation = validateLabResult(formData);
   if (!validation.isValid) {
     console.error(validation.errors);
   }

5. USE FIRESTORE MANAGER
   const results = await labResultManager.getAll(userId);
   await labResultManager.add(userId, labData);
   await labResultManager.update(userId, id, updates);
   await labResultManager.delete(userId, id);
